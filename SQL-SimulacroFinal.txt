	ApePaterno NVARCHAR(100) NOT NULL,
	ApeMaterno NVARCHAR(100) NOT NULL,
	PromedioFinal int not null default 0

GO

CREATE TABLE DBO.NOTAS(
  Id int primary key identity (1,1),
  CodigoAlumno NVARCHAR(20) NOT NULL,
  NombreEvaluacion  NVARCHAR(20) NOT NULL,
  Nota decimal not null default 0
)
GO

CREATE OR ALTER PROCEDURE sp_listar_alumnos 
@filtro NVARCHAR(20) = NULL
AS
BEGIN
SELECT 
	ID, Codigo, Nombres, ApePaterno, ApeMaterno, PromedioFinal
FROM dbo.ALUMNOS AL
WHERE 
	Codigo LIKE '%' + @filtro + '%' 
	OR Nombres LIKE '%' + @filtro + '%' 
	OR ApePaterno LIKE '%' + @filtro + '%' 
	OR ApeMaterno LIKE '%' + @filtro + '%' 
END
GO

exec sp_listar_alumnos I20207606
GO

CREATE OR ALTER PROCEDURE sp_obtener_alumno 
@codigo NVARCHAR(20) = NULL
AS
BEGIN
SELECT 
	AL.ID, Codigo, Nombres, ApePaterno, ApeMaterno, PromedioFinal
FROM dbo.ALUMNOS AL 
WHERE 
	Codigo = @codigo;
END
GO

CREATE OR ALTER PROCEDURE sp_insertar_alumno
@Codigo NVARCHAR(20),
@Nombres NVARCHAR(100),
@ApePaterno NVARCHAR(100),
@ApeMaterno NVARCHAR(100),
@PromedioFinal int
AS
BEGIN
	INSERT INTO Dbo.ALUMNOS
	(Codigo, Nombres, ApePaterno, ApeMaterno, PromedioFinal)
	VALUES
	(@Codigo, @Nombres, @ApePaterno, @ApeMaterno, @PromedioFinal)
END
GO

CREATE OR ALTER PROCEDURE sp_eliminar_alumno
@Codigo NVARCHAR(20)
AS
BEGIN
	DELETE FROM dbo.ALUMNOS
	WHERE 
	Codigo = @Codigo
END
GO

CREATE OR ALTER PROCEDURE sp_insertar_nota
@Codigo NVARCHAR(20),
@NombreEvaluacion NVARCHAR(20),
@Nota DECIMAL
AS
BEGIN
	INSERT INTO Dbo.NOTAS
	(CodigoAlumno, NombreEvaluacion, Nota)
	VALUES
	(@Codigo,  @NombreEvaluacion, @Nota)
END
GO

CREATE OR ALTER PROCEDURE sp_eliminar_notas_por_codigoalumno
@Codigo NVARCHAR(20)
AS
BEGIN
	DELETE FROM dbo.NOTAS
	WHERE
	CodigoAlumno = @Codigo
END
GO


CREATE OR ALTER PROCEDURE sp_listar_notas_por_codigoalumno
@codigo NVARCHAR(20) = NULL
AS
	BEGIN
	SELECT
		ID, CodigoAlumno, NombreEvaluacion, Nota
	FROM dbo.NOTAS N
	WHERE
		CodigoAlumno = @codigo
	END
GO
